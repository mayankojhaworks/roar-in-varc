function doAction(a,el){let c=P.config,s=session(),q=currentQ(),id=q?.id;
 switch(a){

 case 'toggle-chip':changeToggle(el.dataset.kind,el.dataset.value);break;

 case 'clear-target':c.targetId=null;c.count=10;persist();render();break;
 case 'qtype':c.qtype=el.dataset.value;c.targetId=null;persist();render();break;
 case 'preset-count':c.count=Number(el.dataset.value);persist();render();break;

 case 'discard-session':if(s&&['running','paused'].includes(s.state)&&confirm('Discard this unfinished session? Submitted practice answers remain in your progress.')){P.active=null;persist();render();}break;
 case 'open-navigator':openNavigator();break;
 case 'close-drawer':closeNavigator();break;
 case 'jump':closeNavigator();if(s?.state==='finished'){UI.reviewAt=Number(el.dataset.index);UI.page='review';render();}else jump(Number(el.dataset.index));break;
 case 'choose':setDraft(el.dataset.option);break;
 case 'submit':submitAnswer();break;
 case 'clear-draft':if(s&&id){delete s.drafts[id];persist();render()}break;

 case 'next':nextQuestion();break;
 case 'prev':if(s)jump(s.index-1);break;

 case 'note':showNote(el.dataset.id||undefined);break;
 case 'save-note':{let target=UI.noteId;if(target&&BY_ID.has(target)){P.notes[target]=$('#noteTextarea')?.value||'';persist();closeModal();notify('Note saved');if(UI.page==='practice'||UI.page==='review')render()}}break;
 case 'end-session':closeNavigator();confirmEnd();break;
 case 'back-result':UI.page='result';UI.reviewAt=null;render();break;
 case 'review-question':UI.reviewAt=Number(el.dataset.index);UI.page='review';render();window.scrollTo({top:0,behavior:'instant'});break;
 case 'review-prev':UI.reviewAt=Math.max(0,UI.reviewAt-1);render();window.scrollTo({top:0,behavior:'instant'});break;
 case 'review-next':UI.reviewAt=Math.min(s.ids.length-1,UI.reviewAt+1);render();window.scrollTo({top:0,behavior:'instant'});break;
 case 'revise-mistakes':c.mode='incorrect';c.targetId=null;c.years=[];c.slots=[];c.topics=[];c.groups=[];c.count=10;c.duration='none';P.active=null;persist();page('setup');break;
 case 'practice-one':{let target=el.dataset.id;c.mode='focused';c.targetId=target;c.years=[];c.slots=[];c.topics=[];c.groups=[];c.count=1;c.duration='none';P.active=null;persist();page('setup');break}
 case 'bank-prev':UI.bankPage=Math.max(0,UI.bankPage-1);$('#bankRows').innerHTML=bankRows();break;
 case 'bank-next':UI.bankPage++;$('#bankRows').innerHTML=bankRows();break;
 case 'export':exportProgress();break;

 }
}
